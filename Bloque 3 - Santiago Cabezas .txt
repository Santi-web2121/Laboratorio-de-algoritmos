1. using System;


class Program
{
    static void Main()
    {
        int op1 = 0, op2 = 0, op3 = 0;
        string voto;


        do
        {
            Console.WriteLine("1 Pizza");
            Console.WriteLine("2 Hamburguesa");
            Console.WriteLine("3 Empanadas");
            Console.Write("Voto (o terminar): ");
            voto = Console.ReadLine();


            if (voto == "1")
                op1++;
            else if (voto == "2")
                op2++;
            else if (voto == "3")
                op3++;


        } while (voto != "terminar");


        if (op1 > op2 && op1 > op3)
            Console.WriteLine("Gano Pizza");
        else if (op2 > op1 && op2 > op3)
            Console.WriteLine("Gano Hamburguesa");
        else if (op3 > op1 && op3 > op2)
            Console.WriteLine("Gano Empanadas");
        else
            Console.WriteLine("Empate");
    }
}

D) using System;


class Program
{
static void Main()
{
double saldo = 1000;
int opcion;


do
{
Console.WriteLine("Saldo: $" + saldo);
Console.WriteLine("1 Depositar");
Console.WriteLine("2 Retirar");
Console.WriteLine("3 Salir");


opcion = int.Parse(Console.ReadLine());


if (opcion == 1)
{
double monto = double.Parse(Console.ReadLine());
saldo += monto;
}
else if (opcion == 2)
{
double monto = double.Parse(Console.ReadLine());


if (monto <= saldo)
saldo -= monto;
else
Console.WriteLine("Saldo insuficiente");
}


} while (opcion != 3);
}
}

N) using System;


class Program
{
    static void Main()
    {
        string frase;


        do
        {
            frase = Console.ReadLine();


            if (frase != "agusfortnite2008")
            {
                string[] vocales = { "a", "e", "i", "o", "u" };


                for (int i = 0; i < vocales.Length; i++)
                {
                    string resultado = frase;


                    resultado = resultado.Replace("a", vocales[i]);
                    resultado = resultado.Replace("e", vocales[i]);
                    resultado = resultado.Replace("i", vocales[i]);
                    resultado = resultado.Replace("o", vocales[i]);
                    resultado = resultado.Replace("u", vocales[i]);


                    Console.WriteLine(resultado);
                }
            }


        } while (frase != "agusfortnite2008");
    }
}
