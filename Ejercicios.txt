8) using System;

class Program

        {

            string[] numeros = texto.Split(',');

            {

10) using System;

class Program
{
    static void Main()
    {
        try
        {
            Console.Write("Ingrese una contraseña: ");
            string contraseña = Console.ReadLine();

            bool tieneNumero = false;

            foreach (char c in contraseña)
            {
                if (char.IsDigit(c))
                {
                    tieneNumero = true;
                }
            }

            if (contraseña.Length < 8 || !tieneNumero)
            {
                throw new ArgumentException();
            }

            Console.WriteLine("Contraseña válida.");
        }
        catch (ArgumentException)
        {
            Console.WriteLine("La contraseña debe tener al menos 8 caracteres y un número.");
        }
    }
}

14) using System;

class Program
{
    static void Main()
    {
        try
        {
            Console.Write("Ingrese una frase: ");
            string frase = Console.ReadLine();

            Console.WriteLine("Caracteres: " + frase.Length);

            string[] palabras = frase.Split(' ');
            Console.WriteLine("Palabras: " + palabras.Length);

            int vocales = 0;

            foreach (char letra in frase)
            {
                if ("aeiouAEIOU".Contains(letra))
                {
                    vocales++;
                }
            }

            Console.WriteLine("Vocales: " + vocales);

            Console.Write("Ingrese una posición: ");
            int posicion = int.Parse(Console.ReadLine());

            if (posicion < 0 || posicion >= frase.Length)
            {
                throw new IndexOutOfRangeException();
            }

            Console.WriteLine("Carácter: " + frase[posicion]);

            Console.Write("Ingrese un número: ");
            int numero = int.Parse(Console.ReadLine());

            Console.WriteLine("Resultado: " + (100 / numero));
        }
        catch (FormatException)
        {
            Console.WriteLine("Formato incorrecto.");
        }
        catch (IndexOutOfRangeException)
        {
            Console.WriteLine("Posición fuera de rango.");
        }
        catch (DivideByZeroException)
        {
            Console.WriteLine("No se puede dividir por cero.");
        }
        catch (ArgumentException)
        {
            Console.WriteLine("Error de argumento.");
        }
    }
}
