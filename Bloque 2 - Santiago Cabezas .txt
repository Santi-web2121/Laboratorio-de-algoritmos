{
    Console.Write("Nombre: ");
    string nombre = Console.ReadLine();


    for (int i = 0; i < cantidad; i++)
    {
        if (nombres[i] == nombre)
        {
            repetido = true;
        }
    }


    if (!repetido)

        nombres[cantidad] = nombre;
        cantidad++;
    }
}

P) int usuario = 0;
int computadora = 0;


Random r = new Random();


for (int i = 1; i <= 3; i++)
{
    Console.Write("1-Piedra 2-Papel 3-Tijera: ");
    int jugada = int.Parse(Console.ReadLine());


    int pc = r.Next(1, 4);


    if ((jugada == 1 && pc == 3) ||
        (jugada == 2 && pc == 1) ||
        (jugada == 3 && pc == 2))
    {
        usuario++;
    }
    else if (jugada != pc)
    {
        computadora++;
    }
}


if (usuario > computadora)
{
    Console.WriteLine("Ganaste");
}
else if (computadora > usuario)
{
    Console.WriteLine("Ganó la computadora");
}
else
{
    Console.WriteLine("Empate");
}
